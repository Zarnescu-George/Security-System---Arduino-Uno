#include <Servo.h>
#include <IRremote.hpp>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

int led1 = 3;
int led2 = 4;
int led3 = 5;
int buzzer = 8;
int trig = 9;
int echo = 10;
int pinIR = 11;

bool deblocat = false;

Servo servo;

LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  pinMode(led3, OUTPUT);

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("System : OK");

  pinMode(trig, OUTPUT);
  pinMode(echo, INPUT);

  Serial.begin(9600);

  pinMode(buzzer, OUTPUT);

  servo.attach(6);

  IrReceiver.begin(pinIR, ENABLE_LED_FEEDBACK);
}

void loop() {
  if (deblocat) {
    digitalWrite(led1, HIGH);
    digitalWrite(led2, LOW);
    digitalWrite(led3, LOW);
    digitalWrite(buzzer, LOW);
    servo.write(0);

    if (IrReceiver.decode()) {
        unsigned long codPrimit = IrReceiver.decodedIRData.decodedRawData;

        if (codPrimit == 0xE31CFF00)
        {  
            deblocat = false;

            digitalWrite(led1, LOW);
            digitalWrite(led2, LOW);
            digitalWrite(led3, HIGH);

            servo.write(0);
            digitalWrite(buzzer, LOW);

            lcd.clear();
            lcd.setCursor(0,0);
            lcd.print("System : OK");
        }

        IrReceiver.resume();
    }
    return;
}
  digitalWrite(trig, LOW);
  delayMicroseconds(2);

  digitalWrite(trig, HIGH);
  delayMicroseconds(10);
  digitalWrite(trig, LOW);

  long durata = pulseIn(echo, HIGH);
  float distanta = durata * 0.034 / 2;

  if (distanta < 10) {

    digitalWrite(led3, LOW);
    digitalWrite(led2, HIGH);

    delay(500);

    lcd.backlight();
    lcd.setCursor(0, 0);
    lcd.print("System : BLOCKED");

    servo.write(180);

    digitalWrite(buzzer, HIGH);
    delay(50);
    digitalWrite(buzzer, LOW);
    delay(50);
   if (IrReceiver.decode()) {
    unsigned long codPrimit = IrReceiver.decodedIRData.decodedRawData;

    if (codPrimit == 0xE31CFF00) {
    deblocat = true;

    digitalWrite(led1, HIGH);
    digitalWrite(led2, LOW);
    digitalWrite(led3, LOW);

    servo.write(0);

    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("System : UNLOCKED");

}
   else 
   {
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("Wrong Pass");
   }
    IrReceiver.resume();
}
  }
  else {

    digitalWrite(led2, LOW);
    digitalWrite(led3, HIGH);

    delay(500);

    digitalWrite(buzzer, LOW);

    lcd.backlight();
    lcd.setCursor(0, 0);
    lcd.print("System : OK           ");

    servo.write(0);
  }
}
